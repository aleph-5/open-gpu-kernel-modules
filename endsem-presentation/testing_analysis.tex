% -- Benchmark workload descriptions -------------------------------------------
\begin{frame}{Performance Testing}{Benchmark Workload Descriptions}
\vspace{-2mm}
\begin{center}
\scriptsize
\setlength{\tabcolsep}{4pt}
\begin{tabular}{lp{2.8cm}p{6.5cm}}
\toprule
\textbf{Benchmark} & \textbf{Access Pattern} & \textbf{Description} \\
\midrule
\textbf{4K stride}   & Strided writes        & Custom kernel: writes every 4\,KB page once in stride order; stresses worst-case fault rate (every page is a new write fault) \\
\textbf{Seq.}        & Sequential writes     & Same custom kernel with sequential page writes; tests tracking overhead under a linear scan \\
\textbf{GEMM}        & Dense matrix read/write & General matrix multiply (PolyBench); large working set with reuse - most pages written once \\
\textbf{2MM}         & Dense matrix chain    & Two matrix multiplications chained (PolyBench); higher arithmetic intensity than GEMM, fewer write faults per byte \\
\textbf{BICG}        & Sparse-style access   & BiCG linear solver (PolyBench); mixed row/column scans over matrices, irregular write pattern \\
\textbf{MVT}         & Vector-matrix         & Matrix-vector transpose product (PolyBench); two passes over the same matrix, moderate reuse \\
\textbf{Needle}      & DP table fill         & Needleman-Wunsch sequence alignment (Rodinia); diagonal wavefront writes into a 2-D DP table \\
\bottomrule
\end{tabular}

\vspace{1mm}
{\tiny \emph{4K stride} and \emph{Seq.}\ are custom \texttt{int\_set\_uvm} microbenchmarks.
PolyBench and Rodinia workloads use \texttt{cudaMallocManaged} with oversubscription support.}
\end{center}
\end{frame}

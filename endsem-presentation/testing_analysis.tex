% -- Performance: write-intensive graphs ----------------------------------------
\begin{frame}{Performance Testing}{Write-Intensive Benchmarks (\texttt{int\_set\_uvm})}
\begin{center}
\includegraphics[width=0.75\textwidth]{overhead_write.pdf}
\end{center}
\vspace{-5mm}
\begin{itemize}[<+->]
\tiny
\item Overhead scales with \textbf{write-fault density}: 4K-stride hits every page fresh (max fault rate) --- $+57.5\%$ at 512\,MB growing to $+1730.7\%$ at 48\,GB
\item Sequential access coalesces memory traffic $\to$ fewer faults per byte $\to$ \textbf{2--8$\times$ lower} overhead at the same working-set size
\item Oversubscribed regime ($\geq$48\,GB): first-write-wins caps insertions to \textbf{one per page per epoch} regardless of eviction-driven re-faulting
\end{itemize}
\end{frame}

% -- Performance: polybench graphs -----------------------------------------------
\begin{frame}{Performance Testing}{PolyBench Kernels}
\begin{center}
\includegraphics[width=0.7\textwidth]{overhead_polybench.pdf}
\end{center}
\vspace{-3mm}
\begin{itemize}[<+->]
\tiny
\item \textbf{GEMM/2MM}: high arithmetic intensity dominates runtime $\to$ \textbf{13--26\%} overhead while working set fits in VRAM; 2MM lowest of all (9\%) at large sizes due to chained computation
\item \textbf{Regime shift at $\geq$32\,GB}: GEMM baseline drops sharply (205\,s $\to$ 19\,s), making fixed tracking cost appear disproportionately large (207--261\%) --- benchmark-internal, not a tracking effect
\item \textbf{BICG/MVT}: read-heavy but non-trivial overhead from initial page-population faults and small write footprints; amplified further in the oversubscribed regime
\end{itemize}
\end{frame}

% -- Performance: needle graph --------------------------------------------------
\begin{frame}{Performance Testing}{Rodinia Needle - Two-Regime Behavior}

\begin{center}
\includegraphics[width=0.8\textwidth]{overhead_needle.pdf}
\end{center}
\vspace{-1mm}
\begin{itemize}[<+->]
\tiny
\item \textbf{4--16\,GB}: dense DP score-matrix fills every cell $\to$ high first-write fault rate per kernel invocation $\to$ \textbf{871--960\%} overhead
\item \textbf{$\geq$32\,GB}: \texttt{-mb} parameter cuts iteration count drastically, write footprint collapses, overhead drops to \textbf{near zero} ($+0.6$--$+27.8\%$)
\item Confirms \textbf{no fixed overhead floor}: the tracking pipeline adds cost only proportional to actual write-fault work
\end{itemize}
\end{frame}

% -- Performance: query frequency sensitivity -----------------------------------
\begin{frame}{Performance Testing}{Query Frequency Sensitivity}
\begin{center}
\includegraphics[width=0.82\textwidth]{overhead_query_freq.pdf}
\end{center}
\vspace{-1mm}
\begin{itemize}[<+->]
\tiny
\item All three lines (1, 5, or 10 \texttt{query\_dump} calls per interval) are \textbf{nearly coincident} for every benchmark
\item \texttt{query\_dump} in delta mode \textbf{moves a pointer}; it does not re-scan the backing structure proportional to its size, so more epochs add negligible cost
\item Slight spread on Needle reflects \textbf{sleep-interval variability}, not an epoch-count effect
\end{itemize}
\end{frame}


\begin{frame}{Performance Testing}{Theoretical Bounds for Data Structures}
\begin{center}
\renewcommand{\arraystretch}{1.15}
\scriptsize
\begin{tabular}{l l l l p{52mm}}
\toprule
\textbf{Backend} & \textbf{Insert} & \textbf{Lookup} & \textbf{Iterate} & \textbf{Notes} \\
\midrule
bitmap           & $O(1)$            & $O(1)$       & $O(P/w + k)$  & flat \texttt{vzalloc}'d $2^{38}$-bit map; \textbf{no timestamps}; 32\,MB fixed \\
xarray           & $O(1)$ amortised  & $O(1)$       & $O(n)$        & atomic \texttt{xa\_cmpxchg}; timestamps stored \\
sorted vector    & $O(n)$            & $O(\log n)$  & $O(\log n+k)$ & binary-search insert; no internal lock \\
unsorted vector  & $O(n)$            & $O(n)$       & $O(n)$        & append-only; IRQ-safe spinlock \\
linked list      & $O(n)$            & $O(n)$       & $O(n)$        & sentinel head; no extra lock \\
nested bitmap    & $O(1)$            & $O(1)$       & $O(B+k)$      & 2-level xarray+bitmap; timestamps stored \\
chunked vector   & $O(1)$ amortised  & $O(\log n + n_c)$ & $O(n)$    & background \texttt{kthread} sort \\
\bottomrule
\end{tabular}
\end{center}
\vspace{1mm}
{\scriptsize $n=$ dirty pages, $k=$ pages in query range,
$P=2^{38}$, $w=$ word size, $B=$ allocated VA blocks,
$n_c=$ entries in active chunk.}

\vspace{1.5mm}
{\scriptsize
\begin{itemize}
  \item Backends are swappable via the \texttt{struct uvm\_dirty\_ds} abstraction layer. Each backend exposes the same ops vtable (\texttt{init}, \texttt{insert}, \texttt{lookup},
\texttt{for\_each\_in\_range}, \texttt{destroy})
  \item Switching is a one-line change to the static initialiser of \texttt{g\_dirty\_ds}, or done at runtime via \texttt{dirty\_tracking\_hint}.
  \item Inline timing wrappers in \texttt{uvm\_dirty\_ds.h} accumulate per-op latency counters,
gated by \texttt{unlikely(stats.enabled)} so the disabled path adds no measurable overhead.

\end{itemize}
}
\end{frame}
